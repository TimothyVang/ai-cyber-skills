# Commands the git guard blocks. One extended regular expression per line, matched against the
# whole shell command. Lines starting with # and blank lines are ignored. Keep to syntax that both
# grep -E and JavaScript accept: ( ) | [ ] * + ? ^ $ and literal spaces. No \b, \s, or [[:space:]].
# Every pattern starts with the same prefix, which skips git's global options: -c and -C with their value,
# --git-dir, --work-tree and --namespace with theirs, and any other "-option". A value may hold quoted parts
# with spaces (-c user.name="A B", --git-dir="C:/My Repo/.git").
# Arguments between the subcommand and the dangerous flag never cross a shell separator: "( +[^ ;&|]+)*".
# Both guards match one line at a time, after joining lines that end in a backslash.
# A flag or ref ends at a space, a shell separator, a closing parenthesis, or the end: "( |;|&|\||\)|$)".
# A line "@branch <branch regex> <command regex>" blocks the command only while the branch checked out in
# the command's working directory fully matches the branch regex (no spaces in it). Outside a repo or on a
# detached HEAD it never blocks.

# Discards uncommitted work
git( +(-[cC]|--git-dir|--work-tree|--namespace) +("[^"]*"|'[^']*'|[^ "'])+| +-("[^"]*"|'[^']*'|[^ "';&|])+)* +reset( +[^ ;&|]+)* +--hard
git( +(-[cC]|--git-dir|--work-tree|--namespace) +("[^"]*"|'[^']*'|[^ "'])+| +-("[^"]*"|'[^']*'|[^ "';&|])+)* +clean( +[^ ;&|]+)* +(-[a-zA-Z]*f|--force)
git( +(-[cC]|--git-dir|--work-tree|--namespace) +("[^"]*"|'[^']*'|[^ "'])+| +-("[^"]*"|'[^']*'|[^ "';&|])+)* +(checkout|restore)( +-[^ ]+)* +\.( |;|&|\||\)|$)
git( +(-[cC]|--git-dir|--work-tree|--namespace) +("[^"]*"|'[^']*'|[^ "'])+| +-("[^"]*"|'[^']*'|[^ "';&|])+)* +stash +(clear|drop)
git( +(-[cC]|--git-dir|--work-tree|--namespace) +("[^"]*"|'[^']*'|[^ "'])+| +-("[^"]*"|'[^']*'|[^ "';&|])+)* +worktree +remove( +[^ ;&|]+)* +(-f|--force)

# Deletes branches or rewrites history
git( +(-[cC]|--git-dir|--work-tree|--namespace) +("[^"]*"|'[^']*'|[^ "'])+| +-("[^"]*"|'[^']*'|[^ "';&|])+)* +branch( +[^ ;&|]+)* +(-D|--delete +--force|--force +--delete|-d +-f|-f +-d)( |;|&|\||\)|$)
git( +(-[cC]|--git-dir|--work-tree|--namespace) +("[^"]*"|'[^']*'|[^ "'])+| +-("[^"]*"|'[^']*'|[^ "';&|])+)* +(filter-branch|filter-repo)
git( +(-[cC]|--git-dir|--work-tree|--namespace) +("[^"]*"|'[^']*'|[^ "'])+| +-("[^"]*"|'[^']*'|[^ "';&|])+)* +update-ref +-d
git( +(-[cC]|--git-dir|--work-tree|--namespace) +("[^"]*"|'[^']*'|[^ "'])+| +-("[^"]*"|'[^']*'|[^ "';&|])+)* +reflog +expire
git( +(-[cC]|--git-dir|--work-tree|--namespace) +("[^"]*"|'[^']*'|[^ "'])+| +-("[^"]*"|'[^']*'|[^ "';&|])+)* +gc( +[^ ;&|]+)* +--prune=now

# Pushes that overwrite or delete remote history. --force-with-lease to a feature branch stays allowed.
git( +(-[cC]|--git-dir|--work-tree|--namespace) +("[^"]*"|'[^']*'|[^ "'])+| +-("[^"]*"|'[^']*'|[^ "';&|])+)* +push( +[^ ;&|]+)* +(--force|-f)( |;|&|\||\)|$)
git( +(-[cC]|--git-dir|--work-tree|--namespace) +("[^"]*"|'[^']*'|[^ "'])+| +-("[^"]*"|'[^']*'|[^ "';&|])+)* +push( +[^ ;&|]+)* +-[a-zA-Z]*f[a-zA-Z]*( |;|&|\||\)|$)
git( +(-[cC]|--git-dir|--work-tree|--namespace) +("[^"]*"|'[^']*'|[^ "'])+| +-("[^"]*"|'[^']*'|[^ "';&|])+)* +push( +[^ ;&|]+)* +\+[^ ]+
git( +(-[cC]|--git-dir|--work-tree|--namespace) +("[^"]*"|'[^']*'|[^ "'])+| +-("[^"]*"|'[^']*'|[^ "';&|])+)* +push( +[^ ;&|]+)* +(--mirror|--delete|-d|--prune)( |;|&|\||\)|$)
git( +(-[cC]|--git-dir|--work-tree|--namespace) +("[^"]*"|'[^']*'|[^ "'])+| +-("[^"]*"|'[^']*'|[^ "';&|])+)* +push( +[^ ;&|]+)* +:[^ ]+

# Pushes straight to the default branch (merges go through a reviewed PR): a destination that names it,
# --all, and, while it is checked out, a push that names no destination or pushes HEAD.
git( +(-[cC]|--git-dir|--work-tree|--namespace) +("[^"]*"|'[^']*'|[^ "'])+| +-("[^"]*"|'[^']*'|[^ "';&|])+)* +push( +[^ ;&|]+)* +([^ ]*:)?(refs/heads/)?(main|master|trunk)( |;|&|\||\)|$)
git( +(-[cC]|--git-dir|--work-tree|--namespace) +("[^"]*"|'[^']*'|[^ "'])+| +-("[^"]*"|'[^']*'|[^ "';&|])+)* +push( +[^ ;&|]+)* +--all( |;|&|\||\)|$)
@branch (main|master|trunk) git( +(-[cC]|--git-dir|--work-tree|--namespace) +("[^"]*"|'[^']*'|[^ "'])+| +-("[^"]*"|'[^']*'|[^ "';&|])+)* +push( +-[^ ]+)*( +[^- ;&|)][^ ;&|)]*)?( +-[^ ]+)* *(;|&|\||\)|$)
@branch (main|master|trunk) git( +(-[cC]|--git-dir|--work-tree|--namespace) +("[^"]*"|'[^']*'|[^ "'])+| +-("[^"]*"|'[^']*'|[^ "';&|])+)* +push( +[^ ;&|]+)* +(HEAD|@)( |;|&|\||\)|$)
